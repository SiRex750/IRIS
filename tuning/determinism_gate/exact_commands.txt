# --- run_determinism_gate.py ---
import json, sys
from pathlib import Path
REPO = Path("/home/ccbd/IRIS-1")
sys.path.insert(0, str(REPO)); sys.path.insert(0, str(REPO / "scripts"))
import val_confirm_e2e_eval as m
OUT = REPO / "tuning" / "determinism_gate"; OUT.mkdir(parents=True, exist_ok=True)
m.TUNING_DIR = OUT
m.PER_QUESTION_CSV = OUT / "arm_per_question.csv"
m.REPORT_PATH = OUT / "report.md"
m.INDEX_CACHE_DIR = OUT / "index_cache"
raw_log_f = open(OUT / "raw_answers_log.jsonl", "w")
orig_generate = m.aria.generate
def logged_generate(*args, **kwargs):
    raw = orig_generate(*args, **kwargs)
    raw_log_f.write(json.dumps({"raw": raw}) + "\n"); raw_log_f.flush()
    return raw
m.aria.generate = logged_generate
sys.argv = ["val_confirm_e2e_eval.py", "--caption-load", str(REPO / "tuning/blind_ablation/captions_dump.json"), "--repeat", "3"]
m.main()
raw_log_f.close()
# --- end run_determinism_gate.py ---
